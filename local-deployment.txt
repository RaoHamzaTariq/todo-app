# 1. Load images into Minikube (Run this for ALL images once built)
minikube image load todo-backend:v1.0.0
minikube image load todo-frontend:v1.0.0
minikube image load mcp-server:v1.0.0
minikube image load reminder-service:v1.0.0
minikube image load recurring-task-service:v1.0.0
minikube image load audit-log-service:v1.0.0
minikube image load notification-service:v1.0.0

# 2. Deploy using Helm (enabling Dapr and Kafka)
helm dependency update ./helm/todo-app
helm install todo-app ./helm/todo-app \
  --namespace todo-app \
  --create-namespace \
  --set backend.image.tag=v1.0.0 \
  --set frontend.image.tag=v1.0.0 \
  --set mcpServer.image.tag=v1.0.0 \
  --set eventServices.reminderService.image.tag=v1.0.0 \
  --set eventServices.recurringTaskService.image.tag=v1.0.0 \
  --set eventServices.auditLogService.image.tag=v1.0.0 \
  --set eventServices.notificationService.image.tag=v1.0.0
